\section{چکیده و ساختار کلی سیستم}
بر اساس معماری تعیین‌شده جهت اعتبارسنجی عملی الگوریتم‌های کنترلی، این پروپوزال به تشریح فاز ساخت نمونه آزمایشگاهی \lr{(Hardware-in-the-Loop)} یک جرثقیل سقفی با درجات آزادی کاهش‌یافته می‌پردازد. در این فاز، متغیر کنترل‌شونده طول کابل \lr{(Reel Rate)} به منظور تمرکز بر روی دینامیک پایه و کاهش پیچیدگی‌های ساختاری فاز اول حذف گردیده و سیستم به صورت یک مکانیزم استاندارد گهواره-پاندول \lr{(Cart-Pendulum)} با طول بازوی نوسانی ثابت (\lr{$l = l_0$}) طراحی شده است. سیستم محرکه از یک استپر موتور حلقه بسته و انتقال قدرت تسمه-پولی بهره می‌برد.

\section{اهداف دقیق و خروجی‌های مورد انتظار پروتوتایپ}
مطابق با الزامات مهندسی سیستم، دستگاه ساخته‌شده باید به طور دقیق قادر به اجرای سناریوها و تولید خروجی‌های زیر باشد:

\begin{enumerate}
    \item \textbf{ردیابی موقعیت با دقت میلی‌متری \lr{(Positioning)}:} کالسکه باید قادر باشد به هر موقعیت مرجع \lr{$x_{ref}$} در طول ریل (با کورس حرکتی مفید حدود 1 متر) با خطای حالت ماندگار کمتر از 1 میلی‌متر منتقل شده و متوقف گردد.
    \item \textbf{میرایی فعال نوسانات پاندول \lr{(Active Anti-Sway)}:} سیستم باید بتواند با اعمال شتاب/ترمزهای دینامیکی دقیق به کالسکه، زاویه نوسان پاندول (\lr{$\theta$}) را در حین حرکت در محدوده مجاز (مانند \lr{$\pm 10^\circ$}) نگه داشته و ارتعاشات پسماند را در کمتر از 3 ثانیه پس از رسیدن به مقصد به طور کامل خنثی کند (\lr{$\theta \rightarrow 0$}).
    \item \textbf{استخراج لحظه‌ای حالات سیستم \lr{(State Estimation)}:} دستگاه باید موقعیت خطی ارابه (\lr{$x$}) و زاویه پاندول (\lr{$\theta$}) را از طریق انکودرهای تعبیه‌شده با نرخ نمونه‌برداری حداقل 100 هرتز استخراج کرده و به کامپیوتر مرکزی ارسال نماید.
    \item \textbf{اجرای بلادرنگ کنترل‌کننده پیش‌بین \lr{(Real-time NMPC)}:} سیستم سخت‌افزاری باید به عنوان یک پلنت فیزیکی در حلقه کنترل قرار گیرد؛ به طوری که سالور بهینه‌سازی در \lr{MATLAB} فرمان‌های حرکتی (\lr{$u$}) را محاسبه و به درایور موتور تزریق کند.
\end{enumerate}

\section{طراحی مکانیکی}
طراحی مکانیکی سیستم با رویکرد صلبیت بالا و کمترین اصطکاک لغزشی صورت گرفته است. نمای سه‌بعدی طراحی‌شده در نرم‌افزار \lr{SolidWorks} در شکل (1) ارائه شده است.
\begin{figure}[H]
    \centering

    \caption{مدل سه‌بعدی پروتوتایپ جرثقیل سقفی با طول بازوی ثابت (محیط \lr{SolidWorks})}

شاسی اصلی از پروفیل‌های آلومینیوم شیاردار 30 در 30 سنگین تشکیل شده است. حرکت خطی کالسکه بر روی یک ریل مینیاتوری دقیق مدل \lr{MGNR15} به همراه واگن خطی بدون لقی \lr{(Backlash-free)} صورت می‌پذیرد تا دینامیک غیرخطی اصطکاک استاتیک به حداقل برسد. بازوی پاندول از جنس آلومینیوم یا فیبر کربن بوده و در انتهای آن جرم متمرکز 1 کیلوگرمی قرار دارد.

\section{معماری الکترونیکی و ابزار دقیق}
برای دستیابی به اهداف فوق، سیستم به تجهیزات الکترونیکی زیر مجهز شده است:
\begin{itemize}
    \item \textbf{عملگر حرکتی:} یک استپر موتور انکودردار (حلقه بسته) نما 23 با گشتاور \lr{$20\mathrm{\ kg.cm}$} به همراه درایور \lr{HBS57} که تضمین‌کننده اجرای دقیق پروفایل‌های سرعت/شتاب بدون از دست دادن پله \lr{(Step Loss)} است.
    \item \textbf{سنسور زاویه پاندول:} یک انکودر افزایشی با رزولوشن 1024 پالس بر دور متصل به مفصل پاندول که زاویه را با دقت بسیار بالا اندازه‌گیری می‌کند.
    \item \textbf{واحد جمع‌آوری داده و واسط \lr{(DAQ)}:} یک برد \lr{Arduino Mega 2560} وظیفه خواندن پالس‌های انکودر پاندول، خواندن لیمیت‌سوئیچ‌های ایمنی، و برقراری ارتباط سریال با نرم‌افزار \lr{MATLAB/Simulink} جهت دریافت فرمان‌های کنترلی را بر عهده دارد.
\end{itemize}
